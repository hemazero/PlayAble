    <script>
        const container = document.getElementById('game-container');
        const wordDisplay = document.getElementById('word-display');
        const scoreDisplay = document.getElementById('score');
        const scoreLabel = document.getElementById('score-label');
        const langOverlay = document.getElementById('lang-overlay');
        const playOverlay = document.getElementById('play-overlay');
        const modalText = document.getElementById('modal-text');
        const actionText = document.getElementById('action-text');
        const wordContainer = document.getElementById('word-container');
        const scoreBox = document.getElementById('score-box');

        const locales = {
            ar: {
                direction: "rtl",
                fontClass: "font-ar",
                scoreName: "النقاط",
                ready: "استعد",
                instruction: "💡 اتبع لون الكلمة الفعلي.. وليس معناها!",
                tapToPlay: "اضغط في أي مكان للبدء 🕹️",
                gameOver: "انتهت الجولة<br>نقاطك: ",
                retry: "اضغط في أي مكان لإعادة المحاولة 🔄",
                colorList: [
                    { name: "أحمر", hex: 0xff3366, str: "#ff3366" },
                    { name: "أزرق", hex: 0x00ffff, str: "#00ffff" },
                    { name: "أخضر", hex: 0x33ff66, str: "#33ff66" },
                    { name: "أصفر", hex: 0xffcc00, str: "#ffcc00" }
                ]
            },
            en: {
                direction: "ltr",
                fontClass: "font-en",
                scoreName: "Score",
                ready: "READY",
                instruction: "💡 Match the Word COLOR.. Not its meaning!",
                tapToPlay: "TAP ANYWHERE TO PLAY 🕹️",
                gameOver: "GAME OVER!<br>Your Score: ",
                retry: "TAP ANYWHERE TO RETRY 🔄",
                colorList: [
                    { name: "RED", hex: 0xff3366, str: "#ff3366" },
                    { name: "BLUE", hex: 0x00ffff, str: "#00ffff" },
                    { name: "GREEN", hex: 0x33ff66, str: "#33ff66" },
                    { name: "YELLOW", hex: 0xffcc00, str: "#ffcc00" }
                ]
            }
        };

        let currentLang = 'ar';
        let colors = locales.ar.colorList;

        let scene, camera, renderer;
        let playerGroup; 
        let particleSystem; 
        let gates = [];
        let isPlaying = false;
        let score = 0;
        let gameSpeed = 0.55;

        let lastWordIndex = -1;
        let lastColorIndex = -1;
        let correctColorIndex = 0;

        const minX = -3.2;
        const maxX = 3.2;
        let targetX = 0; 
        const gatePositionsX = [-2.4, -0.8, 0.8, 2.4];

        let audioCtx;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playSuccessSound() {
            initAudio();
            if (!audioCtx) return;
            let osc = audioCtx.createOscillator();
            let gain = audioCtx.createGain();
            osc.type = 'triangle'; 
            osc.frequency.setValueAtTime(600, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(950, audioCtx.currentTime + 0.1);
            gain.gain.setValueAtTime(0.07, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.1);
        }

        function playFailSound() {
            initAudio();
            if (!audioCtx) return;
            let osc = audioCtx.createOscillator();
            let gain = audioCtx.createGain();
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(180, audioCtx.currentTime);
            osc.frequency.linearRampToValueAtTime(60, audioCtx.currentTime + 0.3);
            gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.3);
        }

        function selectLanguage(lang) {
            currentLang = lang;
            colors = locales[lang].colorList;
            
            document.documentElement.lang = lang;
            document.getElementById('game-container').dir = locales[lang].direction;
            
            scoreLabel.textContent = locales[lang].scoreName;
            wordDisplay.className = locales[lang].fontClass;
            wordDisplay.textContent = locales[lang].ready;
            
            modalText.className = `instruction-badge ${locales[lang].fontClass}`;
            modalText.innerHTML = locales[lang].instruction;
            
            actionText.className = `tap-to-play ${locales[lang].fontClass}`;
            actionText.textContent = locales[lang].tapToPlay;

            langOverlay.style.display = 'none';
            playOverlay.style.display = 'flex';
            
            // إظهار نصوص اللعبة العلوية بشكل منظم الآن
            wordContainer.style.display = 'block';
            scoreBox.style.display = 'block';

            initAudio();
            init3D();
        }

        function init3D() {
            if (scene) return; 

            scene = new THREE.Scene();
            scene.fog = new THREE.FogExp2(0x0b0f19, 0.05); 

            const aspect = container.clientWidth / container.clientHeight;
            camera = new THREE.PerspectiveCamera(aspect < 1 ? 75 : 60, aspect, 0.1, 1000);
            camera.position.set(0, 2.9, 6.0); 
            camera.lookAt(0, 1.1, -5);

            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setSize(container.clientWidth, container.clientHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(renderer.domElement);

            const ambientLight = new THREE.AmbientLight(0xffffff, 1.1);
            scene.add(ambientLight);

            const particleCount = 90;
            const geometry = new THREE.BufferGeometry();
            const positions = [];
            for (let i = 0; i < particleCount; i++) {
                positions.push((Math.random() - 0.5) * 14); 
                positions.push(Math.random() * 5);          
                positions.push(-Math.random() * 40);        
            }
            geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
            const material = new THREE.PointsMaterial({ color: 0xffffff, size: 0.05, transparent: true, opacity: 0.3 });
            particleSystem = new THREE.Points(geometry, material);
            scene.add(particleSystem);

            playerGroup = new THREE.Group();
            const cubeGeo = new THREE.BoxGeometry(0.82, 0.82, 0.82);
            const cubeMat = new THREE.MeshBasicMaterial({ color: 0x94a3b8 }); 
            const innerCube = new THREE.Mesh(cubeGeo, cubeMat);
            playerGroup.add(innerCube);

            const outlineMat = new THREE.MeshBasicMaterial({ color: 0x000000, side: THREE.BackSide });
            const outlineCube = new THREE.Mesh(cubeGeo, outlineMat);
            outlineCube.scale.set(1.12, 1.12, 1.12); 
            playerGroup.add(outlineCube);

            playerGroup.position.set(0, 0.41, 0);
            scene.add(playerGroup);

            const floorGeo = new THREE.PlaneGeometry(8, 200);
            const floorMat = new THREE.MeshStandardMaterial({ color: 0x0d1220, roughness: 0.7 });
            const floor = new THREE.Mesh(floorGeo, floorMat);
            floor.rotation.x = -Math.PI / 2;
            floor.position.z = -80;
            scene.add(floor);

            animate();
        }

        window.addEventListener('resize', () => {
            if (!renderer) return;
            const width = container.clientWidth;
            const height = container.clientHeight;
            camera.aspect = width / height;
            camera.fov = (width / height) < 1 ? 75 : 60;
            camera.updateProjectionMatrix();
            renderer.setSize(width, height);
        });

        let isDragging = false;
        let previousTouchX = 0;

        window.addEventListener('touchstart', e => {
            if (!isPlaying) return;
            isDragging = true;
            previousTouchX = e.touches[0].clientX;
        });

        window.addEventListener('touchmove', e => {
            if (!isPlaying || !isDragging) return;
            let currentTouchX = e.touches[0].clientX;
            let deltaX = currentTouchX - previousTouchX;
            previousTouchX = currentTouchX;

            targetX += (deltaX / container.clientWidth) * 7.5; 
            targetX = Math.max(minX, Math.min(maxX, targetX));
        });

        window.addEventListener('touchend', () => isDragging = false);

        window.addEventListener('mousedown', e => { if(isPlaying) { isDragging = true; previousTouchX = e.clientX; } });
        window.addEventListener('mousemove', e => {
            if (!isPlaying || !isDragging) return;
            let deltaX = e.clientX - previousTouchX;
            previousTouchX = e.clientX;
            targetX += (deltaX / container.clientWidth) * 7.5;
            targetX = Math.max(minX, Math.min(maxX, targetX));
        });
        window.addEventListener('mouseup', () => isDragging = false);

        function startGame() {
            playOverlay.style.display = 'none';
            score = 0;
            gameSpeed = 0.55;
            targetX = 0;
            playerGroup.position.x = 0;
            lastWordIndex = -1;
            lastColorIndex = -1;
            scoreDisplay.textContent = score;

            gates.forEach(g => scene.remove(g.group));
            gates = [];

            isPlaying = true;
            generateChallenge();
        }

        function generateChallenge() {
            let randomWordIndex;
            let randomColorIndex;

            do { randomWordIndex = Math.floor(Math.random() * colors.length); } while (randomWordIndex === lastWordIndex);
            do { randomColorIndex = Math.floor(Math.random() * colors.length); } while (randomColorIndex === lastColorIndex);

            lastWordIndex = randomWordIndex;
            lastColorIndex = randomColorIndex;
            correctColorIndex = randomColorIndex;

            wordDisplay.textContent = colors[randomWordIndex].name;
            wordDisplay.style.color = colors[correctColorIndex].str;

            let shuffled = [...colors].map((c, idx) => ({...c, originalIndex: idx}));
            shuffled.sort(() => Math.random() - 0.5);

            let gateGroup = new THREE.Group();

            for (let i = 0; i < 4; i++) {
                const gateGeo = new THREE.BoxGeometry(1.4, 1.95, 0.2);
                const gateMat = new THREE.MeshBasicMaterial({ color: shuffled[i].hex });
                const gateMesh = new THREE.Mesh(gateGeo, gateMat);
                gateMesh.position.set(gatePositionsX[i], 0.975, 0);
                
                const borderGeo = new THREE.BoxGeometry(1.44, 1.99, 0.16);
                const borderMat = new THREE.MeshBasicMaterial({ color: 0x000000, side: THREE.BackSide });
                const borderMesh = new THREE.Mesh(borderGeo, borderMat);
                gateMesh.add(borderMesh);

                gateGroup.add(gateMesh);
            }

            gateGroup.position.z = -55; 
            scene.add(gateGroup);

            gates.push({
                group: gateGroup,
                colorOrder: shuffled
            });
        }

        function animate() {
            requestAnimationFrame(animate);

            if (isPlaying) {
                const positions = particleSystem.geometry.attributes.position.array;
                for (let i = 2; i < positions.length; i += 3) {
                    positions[i] += gameSpeed;
                    if (positions[i] > 5) {
                        positions[i] = -40 - Math.random() * 15;
                    }
                }
                particleSystem.geometry.attributes.position.needsUpdate = true;

                playerGroup.position.x += (targetX - playerGroup.position.x) * 0.28;

                for (let i = gates.length - 1; i >= 0; i--) {
                    let gate = gates[i];
                    gate.group.position.z += gameSpeed;

                    if (gate.group.position.z >= -0.2) {
                        let playerCurrentX = playerGroup.position.x;
                        let crossedGateIndex = 0;

                        if (playerCurrentX < -1.6) crossedGateIndex = 0;
                        else if (playerCurrentX < 0) crossedGateIndex = 1;
                        else if (playerCurrentX < 1.6) crossedGateIndex = 2;
                        else crossedGateIndex = 3;

                        let chosenGate = gate.colorOrder[crossedGateIndex];

                        if (chosenGate.originalIndex === correctColorIndex) {
                            score++;
                            scoreDisplay.textContent = score;
                            gameSpeed = Math.min(1.1, gameSpeed + 0.015); 
                            
                            playSuccessSound(); 
                            scene.remove(gate.group);
                            gates.splice(i, 1);
                            generateChallenge();
                        } else {
                            endGame();
                        }
                    }
                }
            }

            renderer.render(scene, camera);
        }

        function endGame() {
            isPlaying = false;ب
            playFailSound(); 
            
            // إظهار الرقم بخلفية سوداء لتباين مثالي وممتاز ومريح للعين
            modalText.innerHTML = `${locales[currentLang].gameOver}<br><span class="score-number-highlight">${score}</span><br><br>${locales[currentLang].instruction}`;
            actionText.textContent = locales[currentLang].retry;
            playOverlay.style.display = 'flex';
        }
    </script>
