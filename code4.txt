        // --- AUDIO ---
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const now = audioCtx.currentTime;

            if (type === 'swipe') {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(180, now);
                osc.frequency.exponentialRampToValueAtTime(500, now + 0.08);
                gain.gain.setValueAtTime(0.2, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.08);
                osc.connect(gain); gain.connect(audioCtx.destination);
                osc.start(now); osc.stop(now + 0.08);
            } else if (type === 'goal') {
                [0, 0.05, 0.1].forEach((delay, idx) => {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'sine';
                    const freqs = [523.25, 659.25, 783.99]; 
                    osc.frequency.setValueAtTime(freqs[idx], now + delay);
                    gain.gain.setValueAtTime(0.15, now + delay);
                    gain.gain.linearRampToValueAtTime(0.01, now + delay + 0.2);
                    osc.connect(gain); gain.connect(audioCtx.destination);
                    osc.start(now + delay); osc.stop(now + delay + 0.2);
                });
            } else if (type === 'gameover') {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(200, now);
                osc.frequency.linearRampToValueAtTime(30, now + 0.3);
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.3);
                osc.connect(gain); gain.connect(audioCtx.destination);
                osc.start(now); osc.stop(now + 0.3);
            } else if (type === 'rush') {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(250, now);
                osc.frequency.exponentialRampToValueAtTime(800, now + 0.4);
                gain.gain.setValueAtTime(0.25, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.4);
                osc.connect(gain); gain.connect(audioCtx.destination);
                osc.start(now); osc.stop(now + 0.4);
            } else if (type === 'smash') {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(150, now);
                osc.frequency.linearRampToValueAtTime(40, now + 0.15);
                gain.gain.setValueAtTime(0.3, now);
                gain.gain.linearRampToValueAtTime(0.01, now + 0.15);
                osc.connect(gain); gain.connect(audioCtx.destination);
                osc.start(now); osc.stop(now + 0.15);
            }
        }

        function createPitchSegment(zOffset) {
            const group = new THREE.Group(); group.position.z = zOffset;
            const isEven = Math.abs(zOffset / 10) % 2 === 0;
            const segmentGeo = new THREE.PlaneGeometry(PITCH_WIDTH, 10);
            const mesh = new THREE.Mesh(segmentGeo, isEven ? pitchMat : pitchStripMat);

        }

        function createStadiumSegment(zOffset) {
            const group = new THREE.Group(); group.position.z = zOffset;

            const leftBoard = new THREE.Mesh(boardGeo, wallMat); leftBoard.position.set(-PITCH_WIDTH/2 - 0.4, 0.55, 0);
            const rightBoard = leftBoard.clone(); rightBoard.position.x = PITCH_WIDTH/2 + 0.4;
            group.add(leftBoard, rightBoard);
            scene.add(group); stadiumSegments.push(group);
        }

        function spawnDefenderWallRow(zOffset) {
            const rowGroup = new THREE.Group(); rowGroup.position.z = zOffset; rowGroup.userData = { passed: false };
            const openLane = Math.floor(Math.random() * 3) - 1; 

            for (let lane = -1; lane <= 1; lane++) {
                if (lane === openLane) continue; 
                const blockGeo = new THREE.BoxGeometry(LANE_WIDTH * 0.96, 2.0, 0.8);
                const wallBlock = new THREE.Mesh(blockGeo, concreteWallMat);
                wallBlock.position.set(lane * LANE_WIDTH, 1.0, 0);
                wallBlock.userData = { isWallBlock: true, shattered: false };
                rowGroup.add(wallBlock);
            }

            const goalGroup = new THREE.Group();
            const postH = 3.2; const postW = PITCH_WIDTH - 1;
            const pSideGeo = new THREE.CylinderGeometry(0.1, 0.1, postH, 6);
            const pTopGeo = new THREE.CylinderGeometry(0.1, 0.1, postW, 6);
            const leftP = new THREE.Mesh(pSideGeo, goalFrameMat); leftP.position.set(-postW/2, postH/2, -1.5); 
            const rightP = leftP.clone(); rightP.position.x = postW/2;
            const topP = new THREE.Mesh(pTopGeo, goalFrameMat); topP.rotation.z = Math.PI/2; topP.position.set(0, postH, -1.5);
            const netGeo = new THREE.BoxGeometry(postW, postH, 1.2);
            const net = new THREE.Mesh(netGeo, goalNetMat); net.position.set(0, postH/2, -0.9);
            goalGroup.add(leftP, rightP, topP, net); rowGroup.add(goalGroup);

            const glowGeo = new THREE.PlaneGeometry(PITCH_WIDTH, 1.8);
            const glowMat = new THREE.MeshBasicMaterial({ color: 0x00ff77, transparent: true, opacity: 0.25, side: THREE.DoubleSide });

            rowGroup.add(glow); goalGlows.push(glow);

            scene.add(rowGroup); defenders.push(rowGroup);
        }

        function spawnParticles(pos, count, colorHex, size, force) {
            const geom = new THREE.BufferGeometry(); const positions = []; const velocities = []; const colors = [];
            const baseColor = new THREE.Color(colorHex);
            for(let i=0; i<count; i++) {
                positions.push(pos.x, pos.y, pos.z);
                velocities.push((Math.random() - 0.5) * force, Math.random() * force * 0.8, (Math.random() - 0.5) * force);
                colors.push(baseColor.r, baseColor.g, baseColor.b);
            }
            geom.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
            const mat = new THREE.PointsMaterial({ size: size, vertexColors: true, transparent: true, opacity: 1.0 });
            const pSystem = new THREE.Points(geom, mat);
            pSystem.userData = { velocities, age: 0, maxAge: 30 };
            scene.add(pSystem); particles.push(pSystem);
        }

        function updateParticles(dt) {
            for(let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i]; const posArr = p.geometry.attributes.position.array; const v = p.userData.velocities;
                p.userData.age++; p.material.opacity = 1.0 - (p.userData.age / p.userData.maxAge);
                for(let j=0; j < posArr.length / 3; j++) {
                    posArr[j*3] += v[j*3] * dt; posArr[j*3+1] += v[j*3+1] * dt; v[j*3+1] -= 25 * dt; posArr[j*3+2] += v[j*3+2] * dt;
                }
                p.geometry.attributes.position.needsUpdate = true;
                if(p.userData.age >= p.userData.maxAge) {
                    scene.remove(p); particles.splice(i, 1);
                }
            }
        }

        function initMap() {
            defenders.forEach(d => scene.remove(d)); pitchSegments.forEach(p => scene.remove(p)); stadiumSegments.forEach(s => scene.remove(s)); particles.forEach(p => scene.remove(p));
            defenders = []; pitchSegments = []; stadiumSegments = []; goalGlows = []; particles = [];
            for (let i = 0; i < 8; i++) { createPitchSegment(-i * 10); createStadiumSegment(-i * 20); }
            for (let i = 0; i < 5; i++) { spawnDefenderWallRow(-25 - (i * DEFENDER_SPACING)); }
        }

        function onTouchStart(e) { const touch = e.touches ? e.touches[0] : e; touchStartX = touch.clientX; touchStartY = touch.clientY; }
        function onTouchEnd(e) {
            if (gameState !== 'PLAYING') return;
            const touch = e.changedTouches ? e.changedTouches[0] : e;
            const diffX = touch.clientX - touchStartX; const diffY = touch.clientY - touchStartY;
            if (Math.abs(diffX) > SWIPE_THRESHOLD || Math.abs(diffY) > SWIPE_THRESHOLD) {
                let oldLane = currentLane;
                if (Math.abs(diffX) > Math.abs(diffY)) { currentLane = diffX > 0 ? Math.min(1, currentLane + 1) : Math.max(-1, currentLane - 1); }
                else { currentLane = 0; }
                if(oldLane !== currentLane) playSound('swipe');
            }
        }

        window.addEventListener('touchstart', onTouchStart, { passive: true }); window.addEventListener('touchend', onTouchEnd, { passive: true });

        function updateEnergyUI() {
            const barFill = document.getElementById('energy-fill');
            if (isGoldRush) {

                barFill.style.width = `${pct}%`;
            } else {
                // Scaled tracking bar logic dynamically tied to current scale requirement
                const pct = (rushGoalCounter / currentRushTarget) * 100;
                barFill.style.width = `${pct}%`;
            }
        }

        function startGame() {
            playSound('swipe'); goals = 0; speedMultiplier = 1; currentLane = 0; ballPos.set(0, ballRadius, 0);
            isGoldRush = false; goldRushTimer = 0; rushGoalCounter = 0;
            currentRushTarget = 10; // Reset requirement on fresh start
            goldAuraMat.opacity = 0;
            scene.background = baseFogColor; scene.fog.color = baseFogColor;
            document.getElementById('score-box').classList.remove('rush-ui');
            document.getElementById('score').innerText = goals; document.getElementById('highscore').innerText = highscore;
            document.getElementById('start-screen').classList.add('hidden'); document.getElementById('gameover-screen').classList.add('hidden');
            updateEnergyUI();
            initMap(); gameState = 'PLAYING';
        }

        function triggerGameOver() {
            gameState = 'GAMEOVER'; shakeDuration = 12; shakeIntensity = 0.35; playSound('gameover');
            spawnParticles(ballPos, 40, 0xff1744, 0.4, 12);
            if (goals > highscore) { highscore = goals; localStorage.setItem('wc_high_goals', highscore); }
            document.getElementById('gameover-screen').classList.remove('hidden');
        }

        function triggerGoal() {
            const splash = document.getElementById('goal-splash'); 
            const splashText = document.getElementById('splash-text');
            
            if (!isGoldRush) {
                rushGoalCounter++;
            }

            // Milestone system now validates against variable scale factor
            if (rushGoalCounter >= currentRushTarget && !isGoldRush) {
                isGoldRush = true;
                goldRushTimer = RUSH_DURATION;
                rushGoalCounter = 0; 
                
                // Increase next milestone dynamically: 10 -> 15 -> 20 -> 25, and so on.
                currentRushTarget += 5; 
                
                playSound('rush');
                
                splashText.innerText = "⚡"; // Display pure lightning, "Goal" text completely avoided
                splash.classList.add('active', 'rush');
                document.getElementById('score-box').classList.add('rush-ui');
            } else {
                // Normal goal behavior remains unchanged
                if (!isGoldRush) {
                    splashText.innerText = "GOAL!";
                    splash.classList.remove('rush');
                    splash.classList.add('active');
                }
            }

            updateEnergyUI();
            shakeDuration = 12; shakeIntensity = 0.15; 
            if(!isGoldRush) playSound('goal');
            
            spawnParticles(ballPos, 40, isGoldRush ? 0xffd600 : 0x00e676, 0.4, 12);
            setTimeout(() => splash.classList.remove('active'), 550);
        }

        const clock = new THREE.Clock();
        function animate() {
            requestAnimationFrame(animate); const dt = Math.min(clock.getDelta(), 0.1); 

            if (gameState === 'PLAYING') {
                if (isGoldRush) {
                    goldRushTimer -= dt;
                    updateEnergyUI();
                    goldAuraMat.opacity = 0.4 + Math.sin(Date.now() * 0.03) * 0.2;
                    
                    scene.background = goldFogColor;
                    scene.fog.color = goldFogColor;

                    if (goldRushTimer <= 0) {
                        isGoldRush = false;
                        goldAuraMat.opacity = 0;
                        scene.background = baseFogColor;
                        scene.fog.color = baseFogColor;
                        document.getElementById('score-box').classList.remove('rush-ui');
                        updateEnergyUI();
                    }
                }

                const targetX = currentLane * LANE_WIDTH;
                ballPos.x += (targetX - ballPos.x) * 20 * dt; 
                
                const baseSpeed = FORWARD_SPEED_START * speedMultiplier;
                const forwardSpeed = isGoldRush ? baseSpeed * 1.75 : baseSpeed;
                
                ballPos.z -= forwardSpeed * dt; ballMesh.position.copy(ballPos);
                ballMesh.rotation.x -= (forwardSpeed / ballRadius) * dt;

                ballShadow.position.set(ballPos.x, 0.02, ballPos.z);

                trailTimer += dt;
                if(trailTimer > 0.015) {
                    if (isGoldRush) {
                        spawnParticles(new THREE.Vector3(ballPos.x + (Math.random()-0.5)*0.3, 0.2, ballPos.z + 0.2), 2, 0xffd600, 0.25, 3);
                    } else {
                        spawnParticles(new THREE.Vector3(ballPos.x, 0.1, ballPos.z + 0.3), 1, 0x144d18, 0.15, 1);
                    }
                    trailTimer = 0;
                }

                defenders.forEach(row => {
                    if (Math.abs(ballPos.z - row.position.z) < 1.0) {
                        row.children.forEach(def => {
                            if (def.userData.isWallBlock && !def.userData.shattered) { 
                                const blockLeftX = def.position.x - (LANE_WIDTH * 0.5); const blockRightX = def.position.x + (LANE_WIDTH * 0.5);
                                if (ballPos.x + ballRadius > blockLeftX && ballPos.x - ballRadius < blockRightX) {
                                    if (isGoldRush) {
                                        def.userData.shattered = true;
                                        def.visible = false;
                                        playSound('smash');
                                        shakeDuration = 8; shakeIntensity = 0.25;
                                        spawnParticles(def.position.clone().setZ(row.position.z), 25, 0xffd600, 0.35, 9);
                                    } else {
                                        triggerGameOver();
                                    }
                                }
                            }
                        });
                    }
                    if (!row.userData.passed && ballPos.z < row.position.z - 1.6) {
                        row.userData.passed = true; goals++; document.getElementById('score').innerText = goals;
                        triggerGoal(); 
                        speedMultiplier += SPEED_INCREMENT * 0.02; 
                    }
                });

                pitchSegments.forEach(seg => { if (ballPos.z < seg.position.z - 20) seg.position.z -= 80; });
                stadiumSegments.forEach(seg => { if (ballPos.z < seg.position.z - 40) seg.position.z -= 160; });
                if (defenders.length > 0 && ballPos.z < defenders[0].position.z - 18) {
                    const oldRow = defenders.shift(); scene.remove(oldRow);
                    spawnDefenderWallRow(defenders[defenders.length - 1].position.z - DEFENDER_SPACING);
                }
                goalGlows.forEach(glow => glow.material.opacity = 0.2 + Math.sin(Date.now() * 0.02) * 0.1);
            }

            updateParticles(dt);
            
            const camZDist = isGoldRush ? 9.8 : 8.5;
            camera.position.set(ballPos.x * 0.35, 5.8, ballPos.z + camZDist); 
            camera.lookAt(new THREE.Vector3(ballPos.x * 0.5, 0.8, ballPos.z - 4));

            if (shakeDuration > 0) {
                camera.position.x += (Math.random() - 0.5) * shakeIntensity;
                camera.position.y += (Math.random() - 0.5) * shakeIntensity;
                shakeDuration--;
            }
            renderer.render(scene, camera);
        }

        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        initMap(); renderer.render(scene, camera); animate();
    </script>
