        // --- مشغل نظام الصوت التفاعلي المدمج ---
        let audioCtx = null;
        function initAudio() {
            try {
                if (!audioCtx) {
                    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                }
                if (audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }
            } catch (e) {}
        }

        function playSound(type) {
            try {
                initAudio();
                if (!audioCtx) return;

                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                if (type === 'jump') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(160, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(400, audioCtx.currentTime + 0.1);
                    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
                    osc.start(); osc.stop(audioCtx.currentTime + 0.1);
                } else if (type === 'die') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(250, audioCtx.currentTime);
                    osc.frequency.linearRampToValueAtTime(50, audioCtx.currentTime + 0.3);
                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
                    osc.start(); osc.stop(audioCtx.currentTime + 0.3);
                } else if (type === 'win') {
                    const now = audioCtx.currentTime;
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(261.6, now);
                    osc.frequency.setValueAtTime(329.6, now + 0.08);
                    osc.frequency.setValueAtTime(392.0, now + 0.16);
                    osc.frequency.setValueAtTime(523.2, now + 0.24);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(); osc.stop(now + 0.4);
                } else if (type === 'troll') {
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(300, audioCtx.currentTime);
                    osc.frequency.linearRampToValueAtTime(80, audioCtx.currentTime + 0.15);
                    gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                    osc.start(); osc.stop(audioCtx.currentTime + 0.15);
                }
            } catch (e) {}
        }

                grid.className = 'stages-grid';

                    const btn = document.createElement('div');

                        btn.className = 'stage-btn';
                        btn.innerText = stageNum;
                        btn.onclick = () => launchStage(stageNum);
                    } else {
                        btn.className = 'stage-btn locked';
                        btn.innerText = '🔒';

                listContainer.appendChild(block);

            document.getElementById('game-container').style.display = 'none';

        function addSpike(x, y, w, h) {
            let s = { x, y, w, h, originalY: y, hidden: false, vx: 0, vy: 0 };
            spikes.push(s);
            return s;

        function checkSpikeCollision(player, spike) {
            if (spike.hidden) return false;
            const padX = spike.w * 0.15;
            const padY = spike.h * 0.15;
            return player.x < (spike.x + spike.w - padX) &&
                   player.x + player.width > (spike.x + padX) &&
                   player.y < (spike.y + spike.h) &&
                   player.y + player.height > (spike.y + padY);
        }

        function endGame() {
            if (gameActive) {
                gameActive = false;
                playSound('die');
                screenShake = 15;
                
                for (let i = 0; i < 20; i++) {
                    particles.push({
                        x: player.x + player.width/2,
                        y: player.y + player.height/2,
                        vx: (Math.random() - 0.5) * 10,
                        vy: (Math.random() - 0.5) * 10,
                        size: Math.random() * 5 + 2,
                        alpha: 1,
                        color: getCurrentWorld().spikeColor
                    });
                }

                setTimeout(() => {
                    deaths++;
                    buildStage(stage);
                }, 500);
            }
        }

        // --- تحديث الفيزياء والحركة المحسنة ---
        function update() {
            if (!gameActive) return;

            if (screenShake > 0) screenShake *= 0.9;

            particles.forEach((p, idx) => {
                p.x += p.vx;
                p.y += p.vy;
                p.alpha -= 0.04;
                if (p.alpha <= 0) particles.splice(idx, 1);
            });

            if (keys.left) {
                player.vx = -player.speed;
                player.faceRight = false;
            } else if (keys.right) {
                player.vx = player.speed;
                player.faceRight = true;
            } else {
                player.vx = 0;
            }

            player.vy += player.gravity;
            if (keys.jump && player.grounded) {
                player.vy = player.jumpForce;
                player.grounded = false;
                playSound('jump');
            }

            player.x += player.vx;
            player.grounded = false;

            if (player.x < 0) player.x = 0;
            if (player.x + player.width > VIEW_WIDTH) player.x = VIEW_WIDTH - player.width;

            // اصطدام X
            platforms.forEach(p => {
                if (p.w === 0 || (p.type === 'normal' && p.visible === false)) return;
                if (player.x < p.x + p.w && player.x + player.width > p.x &&
                    player.y < p.y + p.h && player.y + player.height > p.y) {
                    if (player.vx > 0) player.x = p.x - player.width;
                    if (player.vx < 0) player.x = p.x + p.w;
                }
            });

            player.y += player.vy;

            // اصطدام Y مع تعديلات السقف المعكوس
            platforms.forEach(p => {
                if (p.w === 0 || (p.type === 'normal' && p.visible === false)) return;
                if (player.x < p.x + p.w && player.x + player.width > p.x &&
                    player.y < p.y + p.h && player.y + player.height > p.y) {
                    
                    if (player.gravity > 0) { 
                        if (player.vy > 0) { 
                            player.y = p.y - player.height;
                            player.vy = 0;
                            player.grounded = true;

                            if (p.type === 'bridge' && !p.triggered) {
                                p.triggered = true;
                                p.t_vy = 5;
                                playSound('troll');
                            }
                        } else if (player.vy < 0) { 
                            player.y = p.y + p.h;
                            player.vy = 0;
                        }
                    } else { 
                        if (player.vy < 0) { 
                            player.y = p.y + p.h;
                            player.vy = 0;
                            player.grounded = true; 
                        } else if (player.vy > 0) { 
                            player.y = p.y - player.height;
                            player.vy = 0;
                        }
                    }
                }
            });

            const isMoving = Math.abs(player.vx) > 0.1 || Math.abs(player.vy) > 0.1;
            if (isMoving) {
                player.trail.push({ x: player.x, y: player.y });
                if (player.trail.length > 4) player.trail.shift();
                eyeAnimTimer += 0.15;
            } else {
                player.trail = []; 
                eyeAnimTimer = 0; 
            }

            platforms.forEach(p => {
                if (p.type === 'bridge' && p.triggered) {
                    p.y += p.t_vy;
                }
            });

            traps.forEach(t => t.update());

            spikes.forEach(s => {
                if (checkSpikeCollision(player, s)) {
                    endGame();
                }
            });

            if (player.gravity > 0 && player.y > VIEW_HEIGHT) {
                endGame();
            } else if (player.gravity < 0 && player.y + player.height < 0) {
                endGame();
            }

            // الفوز وتجاوز المرحلة والتحكم في نهاية اللعبة
            if (player.x < portal.x + portal.w && player.x + player.width > portal.x &&
                player.y < portal.y + portal.h && player.y + player.height > portal.y) {
                
                gameActive = false; 
                playSound('win');
                
                const banner = document.getElementById('clear-banner');
                const bannerTitle = document.getElementById('clear-title');

                if (stage === 25) {
                    // رسالة النهاية المطلوبة في آخر حلقة
                    bannerTitle.innerHTML = `تهانينا! لقد أنهيت اللعبة بالكامل!<br><span style="font-size: 1.2rem; color: #ff2a5f; text-shadow: 0 0 10px rgba(255,42,95,0.4);">تابعني لأجزاء من اللعبة قادمة</span>`;
                    banner.style.display = 'flex';
                    
                    setTimeout(() => {
                        banner.style.display = 'none';
                        stage = 1;
                        buildMenu();
                    }, 4500);
                } else {
                    bannerTitle.innerText = `تم تجاوز المرحلة ${stage}!`;
                    banner.style.display = 'flex';

                    let isWorldFinished = (stage % 5 === 0); 

                    if (stage === maxUnlockedStage && maxUnlockedStage < 25) {
                        maxUnlockedStage++;
                        saveProgress();
                    }

                    setTimeout(() => {
                        banner.style.display = 'none';
                        
                        if (isWorldFinished) {
                            if (stage < 25) {
                                stage++;
                            }
                            buildMenu(); 
                        } else {
                            if (stage < 25) {
                                stage++;
                                buildStage(stage);
                            }
                        }
                    }, 1500);
                }
            }

            portalRotation += 0.04;
        }

        // --- الرسم والعرض البصري ---
        function draw() {
            if (!gameActive) return;

            const world = getCurrentWorld();
            ctx.fillStyle = world.bg;
            ctx.fillRect(0, 0, VIEW_WIDTH, VIEW_HEIGHT);

            ctx.save();
            if (screenShake > 0) {
                ctx.translate((Math.random() - 0.5) * screenShake, (Math.random() - 0.5) * screenShake);
            }

            traps.forEach(t => { if (t.draw) t.draw(); });

            // رسم الأرضيات
            platforms.forEach(p => {
                if (p.w === 0 || (p.type === 'normal' && p.visible === false)) return;
                ctx.fillStyle = world.platformColor;
                ctx.fillRect(p.x, p.y, p.w, p.h);
                ctx.fillStyle = world.platTop;
                ctx.fillRect(p.x, p.y, p.w, 6);
            });

            // رسم ذيل الحركة (Neon Trail)
            player.trail.forEach((t, index) => {
                ctx.save();
                ctx.globalAlpha = (index + 1) * 0.12;
                ctx.fillStyle = '#00f0ff';
                ctx.fillRect(t.x, t.y, player.width, player.height);
                ctx.restore();
            });

            // رسم اللاعب وعيونه
            if (gameActive) {
                ctx.fillStyle = '#00f0ff';
                ctx.fillRect(player.x, player.y, player.width, player.height);
                ctx.fillStyle = '#000';
                let eyeOffset = player.faceRight ? 12 : 2;
                let bounceY = Math.sin(eyeAnimTimer) * 1.5;
                ctx.fillRect(player.x + eyeOffset, player.y + 6 + bounceY, 4, 6);
                ctx.fillRect(player.x + eyeOffset + 6, player.y + 6 + bounceY, 4, 6);
            }

            // رسم الأشواك
            spikes.forEach(s => {
                if (s.hidden) return;
                ctx.fillStyle = world.spikeColor;
                ctx.beginPath();
                ctx.moveTo(s.x, s.y + s.h);
                ctx.lineTo(s.x + s.w / 2, s.y);
                ctx.lineTo(s.x + s.w, s.y + s.h);
                ctx.closePath();
                ctx.fill();
            });

            // رسم البوابة
            ctx.save();
            ctx.translate(portal.x + portal.w / 2, portal.y + portal.h / 2);
            ctx.rotate(portalRotation);
            ctx.fillStyle = '#33cc66';
            ctx.beginPath();
            ctx.ellipse(0, 0, portal.w / 2, portal.h / 2, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();

            // جزيئات الموت
            particles.forEach(p => {
                ctx.save();
                ctx.globalAlpha = p.alpha;
                ctx.fillStyle = p.color;
                ctx.fillRect(p.x, p.y, p.size, p.size);
                ctx.restore();
            });

            ctx.restore();
        }

        function gameLoop() {
            update();
            draw();
            requestAnimationFrame(gameLoop);
        }

        // --- تهيئة أزرار التحكم اللمسية والفيزيائية ---
        function setupControls() {
            // دالة موحدة لربط أحداث اللمس والماوس لمنع التعليق
            function bindControl(btnId, action) {
                const btn = document.getElementById(btnId);
                
                // اللمس للهواتف
                btn.addEventListener('touchstart', (e) => {
                    e.preventDefault();
                    initAudio();
                    keys[action] = true;
                }, { passive: false });
                
                btn.addEventListener('touchend', (e) => {
                    e.preventDefault();
                    keys[action] = false;
                }, { passive: false });

                btn.addEventListener('touchcancel', (e) => {
                    e.preventDefault();
                    keys[action] = false;
                }, { passive: false });

                // الماوس للحواسيب
                btn.addEventListener('mousedown', (e) => {
                    e.preventDefault();
                    initAudio();
                    keys[action] = true;
                });
                btn.addEventListener('mouseup', (e) => {
                    e.preventDefault();
                    keys[action] = false;
                });
                btn.addEventListener('mouseleave', () => {
                    keys[action] = false;
                });
            }

            bindControl('btn-left', 'left');
            bindControl('btn-right', 'right');
            bindControl('btn-jump', 'jump');

            // دعم لوحة المفاتيح
            window.addEventListener('keydown', (e) => {
                initAudio();
                if (e.key === 'ArrowLeft' || e.key === 'a') keys.left = true;
                if (e.key === 'ArrowRight' || e.key === 'd') keys.right = true;
                if (e.key === 'ArrowUp' || e.key === 'w' || e.key === ' ') keys.jump = true;
            });
            window.addEventListener('keyup', (e) => {
                if (e.key === 'ArrowLeft' || e.key === 'a') keys.left = false;
                if (e.key === 'ArrowRight' || e.key === 'd') keys.right = false;
                if (e.key === 'ArrowUp' || e.key === 'w' || e.key === ' ') keys.jump = false;
            });
        }

        setupControls();
        buildMenu();
        gameLoop();
    </script>
